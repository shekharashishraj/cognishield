Here’s a clean **LangChain Phase 1 skeleton** for CogniShield using:

* **one LLM**
* **different role prompts**
* **Pydantic structured outputs**
* **Python-controlled verifier loop**
* **max 3 revisions**

This directly reflects the paper’s separation of **meta-assistant**, **generator**, **validator suite**, and **feedback verifier** inside a closed-loop refinement process.

# 1) `schemas.py`

```python id="lcskel02"
from typing import List, Optional, Literal, Dict, Any
from pydantic import BaseModel, Field

InterventionType = Literal["scaffold", "hint", "redirect", "defer", "refuse"]


class PlannerOutput(BaseModel):
    intervention: InterventionType
    policy_rationale: str
    generator_instruction: str


class GeneratorOutput(BaseModel):
    response_text: str
    self_check: str


class ValidatorOutput(BaseModel):
    validator_name: str
    score: int = Field(ge=1, le=5)
    passed: bool
    issues: List[str] = Field(default_factory=list)
    fix_suggestion: str = "none"

    answer_leakage: Optional[bool] = None
    direct_solution: Optional[bool] = None
    prompt_injection_detected: Optional[bool] = None
    level_match: Optional[bool] = None
    does_reasoning_for_student: Optional[bool] = None
    factual_risk: Optional[bool] = None


class VerifierDecision(BaseModel):
    decision: Literal["accept", "revise"]
    reasons: List[str] = Field(default_factory=list)
    backprompt: Optional[str] = None


class TurnContext(BaseModel):
    user_query: str
    history: List[Dict[str, str]] = Field(default_factory=list)
    learner_profile: Dict[str, Any] = Field(default_factory=dict)
    rubric_constraints: Dict[str, Any] = Field(default_factory=dict)
    task_context: Dict[str, Any] = Field(default_factory=dict)


class CogniShieldState(BaseModel):
    context: TurnContext
    plan: Optional[PlannerOutput] = None
    candidate: Optional[GeneratorOutput] = None
    validator_reports: Dict[str, ValidatorOutput] = Field(default_factory=dict)
    backprompt: Optional[str] = None
    attempt: int = 0
```

Your job is to choose the safest and most pedagogically appropriate intervention for the learner request.

Allowed interventions:
- scaffold
- hint
- redirect
- defer
- refuse

```python id="lcskel10"
from pathlib import Path
from langchain_core.prompts import ChatPromptTemplate
from app.schemas import PlannerOutput
from app.chains.llm_factory import planner_llm

PROMPT_PATH = Path(__file__).resolve().parent.parent / "prompts" / "planner.txt"

def load_prompt() -> str:
    return PROMPT_PATH.read_text(encoding="utf-8")

def build_planner_chain():
    system_prompt = load_prompt()

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", system_prompt),
            ("human",
             "User query: {user_query}\n"
             "History: {history}\n"
             "Learner profile: {learner_profile}\n"
             "Rubric constraints: {rubric_constraints}\n"
             "Task context: {task_context}\n"),
        ]
    )

    llm = planner_llm().with_structured_output(PlannerOutput)
    return prompt | llm
```

# 10) `verifier.py`

Keep this mostly rule-based in Phase 1.

```python id="lcskel13"
from app.schemas import PlannerOutput, GeneratorOutput, ValidatorOutput, VerifierDecision

def verify_with_rules(
    plan: PlannerOutput,
    candidate: GeneratorOutput,
    reports: dict[str, ValidatorOutput],
) -> VerifierDecision:
    reasons = []

    bloom = reports["bloom"]
    cognitive = reports["cognitive"]
    safety = reports["safety"]
    accuracy = reports["accuracy"]

    if safety.answer_leakage:
        reasons.append("Answer leakage detected.")
    if safety.direct_solution:
        reasons.append("Response gives too direct a solution.")
    if safety.prompt_injection_detected:
        reasons.append("Prompt injection or policy bypass risk detected.")
    if accuracy.score < 4:
        reasons.append("Accuracy too low.")
    if cognitive.score < 3:
        reasons.append("Cognitive engagement too low.")
    if bloom.score < 3:
        reasons.append("Pedagogical level mismatch.")

    if reasons:
        return VerifierDecision(
            decision="revise",
            reasons=reasons,
            backprompt=(
                "Revise the answer to remove answer-level disclosure, increase scaffolding, "
                "preserve factual correctness, and require more student reasoning."
            ),
        )

    return VerifierDecision(
        decision="accept",
        reasons=["All verifier thresholds passed."],
        backprompt=None,
    )
```

# 11) `orchestrator.py`

This is the heart of the prototype.

```python id="lcskel14"
from app.config import MAX_REVISIONS
from app.schemas import CogniShieldState
from app.verifier import verify_with_rules
from app.chains.planner_chain import build_planner_chain
from app.chains.generator_chain import build_generator_chain
from app.chains.validator_chains import (
    build_bloom_validator_chain,
    build_cognitive_validator_chain,
    build_safety_validator_chain,
    build_accuracy_validator_chain,
)

planner_chain = build_planner_chain()
generator_chain = build_generator_chain()
bloom_chain = build_bloom_validator_chain()
cognitive_chain = build_cognitive_validator_chain()
safety_chain = build_safety_validator_chain()
accuracy_chain = build_accuracy_validator_chain()


def run_turn(state: CogniShieldState) -> str:
    ctx = state.context

    state.plan = planner_chain.invoke(
        {
            "user_query": ctx.user_query,
            "history": ctx.history,
            "learner_profile": ctx.learner_profile,
            "rubric_constraints": ctx.rubric_constraints,
            "task_context": ctx.task_context,
        }
    )

    previous_candidate = ""
    backprompt = ""

    for attempt in range(MAX_REVISIONS):
        state.attempt = attempt + 1

        state.candidate = generator_chain.invoke(
            {
                "user_query": ctx.user_query,
                "history": ctx.history,
                "intervention": state.plan.intervention,
                "policy_rationale": state.plan.policy_rationale,
                "generator_instruction": state.plan.generator_instruction,
                "previous_candidate": previous_candidate,
                "backprompt": backprompt,
            }
        )

        validator_payload = {
            "user_query": ctx.user_query,
            "history": ctx.history,
            "intervention": state.plan.intervention,
            "generator_instruction": state.plan.generator_instruction,
            "candidate_response": state.candidate.response_text,
            "rubric_constraints": ctx.rubric_constraints,
            "task_context": ctx.task_context,
        }

        reports = {
            "bloom": bloom_chain.invoke(validator_payload),
            "cognitive": cognitive_chain.invoke(validator_payload),
            "safety": safety_chain.invoke(validator_payload),
            "accuracy": accuracy_chain.invoke(validator_payload),
        }
        state.validator_reports = reports

        verdict = verify_with_rules(state.plan, state.candidate, reports)
        if verdict.decision == "accept":
            return state.candidate.response_text

        previous_candidate = state.candidate.response_text
        backprompt = verdict.backprompt or ""

    return (
        "I can help you work through this step by step. "
        "Start by telling me what you’ve tried so far, and I’ll guide you with a hint."
    )
```

# 15) Why this structure is good for CogniShield

Because it mirrors the paper’s actual logic instead of turning everything into a vague “agent swarm”:

* **Planner** = intervention selection
* **Generator** = response realization
* **Validators** = Bloom, cognitive, safety, accuracy
* **Verifier** = accept or BackPrompt revise
* **Loop** = fixed refinement budget of 2–3 attempts 

# 16) What I’d improve next after this works

After this MVP is stable, the next upgrades should be:

1. add **LangSmith tracing**
2. log every plan, candidate, validator score, and backprompt
3. move prompt text into versioned files
4. add a small test set of:

   * legitimate
   * ambiguous
   * disallowed
5. later replace some rule-based checks with trained validators

That also matches the paper’s staged framing where validators are calibrated and the system is later tuned more systematically.
